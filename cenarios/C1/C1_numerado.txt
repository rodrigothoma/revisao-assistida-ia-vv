 1 | /**
 2 |  * Calcula a tarifa do estacionamento de um shopping.
 3 |  * Todos os valores são em centavos (800 significa R$ 8,00).
 4 |  */
 5 | public class CalculadoraEstacionamento {
 6 | 
 7 |     private static final int TARIFA_PRIMEIRA_HORA = 800;
 8 |     private static final int TARIFA_HORA_ADICIONAL = 300;
 9 | 
10 |     /**
11 |      * Calcula a tarifa a pagar por uma permanência.
12 |      *
13 |      * <p>Regras:</p>
14 |      * <ol>
15 |      *   <li>Permanências de até 15 minutos, inclusive, são gratuitas.</li>
16 |      *   <li>Acima de 15 e até 60 minutos, inclusive, a tarifa é 800.</li>
17 |      *   <li>Acima de 60 minutos, a tarifa é 800 mais 300 por hora adicional
18 |      *       ou fração de hora.</li>
19 |      *   <li>A tarifa de uma permanência nunca passa de 3000.</li>
20 |      *   <li>Ticket validado por loja parceira paga metade da tarifa,
21 |      *       arredondada para baixo. O desconto é aplicado depois do
22 |      *       limite da regra 4.</li>
23 |      * </ol>
24 |      *
25 |      * @param minutos  tempo de permanência, em minutos
26 |      * @param validado {@code true} se o ticket foi validado por loja parceira
27 |      * @return a tarifa, em centavos
28 |      * @throws IllegalArgumentException se {@code minutos} for negativo
29 |      */
30 |     public int calcularTarifa(int minutos, boolean validado) {
31 |         if (minutos < 0) {
32 |             throw new IllegalArgumentException("minutos negativo: " + minutos);
33 |         }
34 |         int tarifa = tarifaPorTempo(minutos);
35 |         if (validado) {
36 |             tarifa = tarifa / 2;
37 |         }
38 |         return tarifa;
39 |     }
40 | 
41 |     /**
42 |      * Calcula a tarifa pelas regras 1 a 3 de
43 |      * {@link #calcularTarifa(int, boolean)}, sem limite e sem desconto.
44 |      *
45 |      * @param minutos tempo de permanência, em minutos, não negativo
46 |      * @return a tarifa, em centavos
47 |      */
48 |     private int tarifaPorTempo(int minutos) {
49 |         if (minutos < 15) {
50 |             return 0;
51 |         }
52 |         if (minutos <= 60) {
53 |             return TARIFA_PRIMEIRA_HORA;
54 |         }
55 |         int horas = horasAdicionais(minutos - 60);
56 |         return TARIFA_PRIMEIRA_HORA + horas * TARIFA_HORA_ADICIONAL;
57 |     }
58 | 
59 |     /**
60 |      * Converte os minutos além da primeira hora em horas adicionais,
61 |      * contando a fração de hora como hora inteira.
62 |      *
63 |      * @param minutosExcedentes minutos além da primeira hora, maior que zero
64 |      * @return o número de horas adicionais
65 |      */
66 |     private int horasAdicionais(int minutosExcedentes) {
67 |         return (minutosExcedentes + 59) / 60;
68 |     }
69 | }
